\recipe{CrumbJacks}
\label{recipe:crumble_flapjacks}
\makes{12 portions}
\worktime{\SI{20} \minute}
\totaltime{\SI{50} \minute}
\source{\href{https://tinandthyme.uk/2021/09/blackberry-apple-crumble-bars/}{tinandthyme.uk/2021/09/blackberry-apple-crumble-bars/}}

\begin{ingredients}
  \ingroup{Base \& Topping}
  \item \SI{200}{\gram} Plain Flour
  \item \SI{200}{\gram} Porridge Oats
  \item \SI{1}{\teaspoon} Baking Powder
  \item \SI{100}{\gram} Butter
  \item \SI{100}{\gram} Brown Sugar
  \item Optional: 1 squirt of Syrup
  
  \ingroup{Filling}
  \item \SI{200}{\gram} Frozen Red Fruits

\end{ingredients}

\begin{method}
  	Set the oven to \SI{180}{\degreeCelsius}.
  	
	In a large bowl, mix the flour, oats, and baking powder together.
	
	Melt the butter with the brown sugar in a sauce pan.
	
	Heat and mash the red fruit filling in another sauce pan.
	
	Mix the butter \& sugar with the flour \& oats to make the base \& topping.
	
	Press half the base/topping mixture into a 20 cm square tin or pyrex dish.
	
	Add the red fruit filling on top.
	
	Scatter the remainder of the base/topping mixture on top, trying not to mix with the filling.

	Bake for 30 minutes until golden. 
	
	Let cool on a wire rack before cutting into twelve bars.

\end{method}
