\recipe{Fake Gu Pots}
\label{recipe:fake_gu_pots}
\makes{6 portions}
\worktime{\SI{40} \minute}
\totaltime{\SI{40} \minute}
\source{Bea}


\begin{ingredients}

  \item double cream
  \item toffee sweets
  \item Digestive biscuits or shortbread
  \item Butter
\end{ingredients}


\begin{method}
To make the Base
Crush the digestive biscuits until they are smaller than 0.5cm.

Melt some butter in a medium sized bowl.

Mix the Biscuits and butter.

Firmly press the biscuit mix into short glass jars.

Cool in the fridge for 10min.

To make the caramel
Melt the sweets in a sauce pan with the double cream.

Pour into the jars with the biscuit base and return to the fridge for
10min,

To make the Chocolate topping 
Melt the Chocolate with the double cream.

Pour into the jars and return to the fridge for 10 min.
  
\end{method}
