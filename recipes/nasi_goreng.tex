\recipe{Nasi Goreng}
\label{recipe:nasi_goreng}
\makes{1 portion}
\worktime{\SI{20} \minute}
\totaltime{\SI{30} \minute}
\source{Bea}


\begin{ingredients}

  \item 2 tbsp vegetable oil
  \item 1 small onion finely sliced
  \item 2 garlic cloves crushed
  \item 1 carrot grated
  \item $1/2$ small Chinese or Savoy cabbage shredded
  \item 175g cooked brown rice
  \item 1 tbsp fish sauce (optional)
  \item 1 tbsp soy sauce
  \item 1 egg
  \item hot sriracha
  \item chilli sauce, to serve
\end{ingredients}


\begin{method}
Heat the oil in a wok over a high heat. Add the onion and cook for 3-4
mins until softened and slightly caramelised. Add the garlic and stir
for 1 min.

Toss in the carrot and cabbage, then cook for 1-2 mins. Add the rice
and stir to warm through. Pour in the fish sauce, soy sauce and some
seasoning. Make a well in the centre of the wok and crack in the egg.
Fry until the white is nearly set.

Serve the rice in a large bowl, topped with the fried egg and drizzled
with chilli sauce.
  
\end{method}
