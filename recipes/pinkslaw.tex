\recipe{Pinkslaw}
\label{recipe:pinkslaw}
\makes{6 Side portions}
\worktime{\SI{10} \minute}
\totaltime{\SI{10} \minute}
\source{Murray}


\begin{ingredients}

  \item $1/2$ small red cabbage (finely sliced)
  \item 1 small onion (thinly sliced)
  \item 1 large carrot (shredded)
  \item 1 large celery stalk (finely diced)
  \item \qty{0.75}{\cup} mayonnaise
  \item \qty{0.5}{\teaspoon} celery salt
  \item \qty{25}{\teaspoon} black pepper
  \item \qty{1}{\tablespoon} apple cider vinegar
  \item \qty{2}{\tablespoon} lemon juice
\end{ingredients}


\begin{method}
  	In a large mixing bowl, combine all the ingredients and give
everything a good mix.

You can enjoy it right away or chill it before serving
  
\end{method}
