\recipe{Quick \& Easy Red Lentil Dahl}
\label{recipe:red_lentil_dahl}
\makes{8 portions}
\worktime{\SI{30}{\minute}}
\totaltime{\SI{50}{\minute}}
\source{\href{https://www.noracooks.com/red-lentil-dahl/}{noracooks.com/red-lentil-dahl}}

\begin{ingredients}
\item \SI{1}{\tablespoon} olive oil
\item \SI{1} large yellow onion, chopped small
\item \SI{5} cloves garlic, minced
\item \SI{1}{\tablespoon} fresh ginger, peeled and grated
\item \SI{1}{\tablespoon} garam masala
\item \SI{1}{\teaspoon} ground turmeric
\item \SI{0.5}{\teaspoon} red pepper chili flakes
\item \SI{270}{\gram} dried red lentils
\item \SI{397}{\milli\litre} canned diced tomatoes
\item \SI{383}{\milli\litre} full-fat coconut milk
\item \SI{705}{\milli\litre} vegetable broth
\item \SI{1}{\teaspoon} salt, or to taste
\item juice of \SI{0.5} lemon
\item \SI{90}{\gram} baby spinach
\end{ingredients}

\begin{method}
Heat \SI{1}{tablespoon} olive oil in a large pot or pan over medium heat. Add the chopped onion and sauté for \SI{5}{\minute}, stirring frequently.

Add \SI{5} cloves garlic and \SI{1}{tablespoon} grated ginger. Cook for \SI{1}{\minute}, until fragrant.

Add \SI{1}{tablespoon} garam masala, \SI{1}{teaspoon} turmeric and \SI{0.5}{\teaspoon} red pepper chili flakes. Stir into the onion mixture. If the mixture is too dry, add a few tablespoons of water.

Add \SI{270}{\gram} dried red lentils, \SI{397}{\milli\litre} canned diced tomatoes with their juices, \SI{383}{\milli\litre} coconut milk and \SI{705}{\milli\litre} vegetable broth. Stir well and turn the heat to high. Bring to a boil, then reduce the heat and simmer for about \SI{15}{\minute}, stirring occasionally, until the lentils are soft and cooked.

Squeeze the juice of \SI{0.5} lemon into the pan and stir in \SI{90}{\gram} baby spinach until wilted. Season with \SI{1}{\teaspoon} salt, or to taste.


\end{method}

\begin{notes}
Serve with cooked rice and vegan naan. Drizzle with a little full-fat coconut milk or plain vegan yogurt, or serve with cashew cream.

For a coconut-free version, cashew cream can be used instead.
\end{notes}
